```rust
//! Series-scoped deletes for the Google Calendar event cache (specs/0054 W5).
//!
//! Lives outside `google_calendar.rs` because that file is pinned by the
//! specs/0042 file-size ratchet and may only shrink.

use sqlx::{Error as SqlxError, SqlitePool};

use super::google_calendar::GoogleCalendarRepository;

impl GoogleCalendarRepository {
    /// Delete every cached row belonging to one recurring series — the master
    /// row AND all expanded instances (`gcal:<cal>/<series>` and
    /// `gcal:<cal>/<series>_…`). Returns the number of rows removed.
    ///
    /// `LIKE` metacharacters in the id are escaped: Google event ids are
    /// base32hex and calendar ids are email-shaped, so neither should contain
    /// `%` or `_`… except instance ids, which are literally `<id>_<ts>`. The
    /// prefix here is the SERIES id, but escape anyway rather than rely on it.
    pub async fn delete_events_for_series(
        pool: &SqlitePool,
        calendar_id: &str,
        series_id: &str,
    ) -> Result<u64, SqlxError> {
        let prefix = format!("gcal:{calendar_id}/{series_id}");
        let escaped = prefix.replace('\\', r"\\").replace('%', r"\%").replace('_', r"\_");
        let result = sqlx::query(
            r"DELETE FROM google_calendar_events
              WHERE calendar_id = ?
                AND (id = ? OR id LIKE ? ESCAPE '\')",
        )
        .bind(calendar_id)
        .bind(&prefix)
        .bind(format!("{escaped}\\_%"))
        .execute(pool)
        .await?;
        Ok(result.rows_affected())
    }
}
```

/// Max days of consecutive incremental syncs before a full resync is forced
/// (specs/0054 W5). The horizon check alone allows ~53 days of drift.
const MAX_INCREMENTAL_DAYS: i64 = 7;

/// True while incremental syncs remain trustworthy: the window horizon is still
/// comfortably ahead AND the last full/incremental pass is recent enough that
/// cache drift cannot accumulate.
fn incremental_is_safe(
    window_ends_at: Option<&str>,
    last_synced_at: Option<&str>,
    now: DateTime<Utc>,
) -> bool {
    horizon_healthy(window_ends_at, now)
        && last_synced_at
            .and_then(|t| t.parse::<DateTime<Utc>>().ok())
            .is_some_and(|t| now - t < Duration::days(MAX_INCREMENTAL_DAYS))
}
